ชุดที่ 5: Oracle Joins & Aggregates

SELECT e.name, d.dept_name FROM emp e JOIN dept d ON e.dept_id = d.id;

SELECT dept_id, AVG(salary) FROM employees GROUP BY dept_id;

SELECT status, COUNT(1) FROM orders GROUP BY status HAVING COUNT(1) > 5;

SELECT p.product_name, c.category_name FROM products p LEFT JOIN categories c ON p.cat_id = c.id;
